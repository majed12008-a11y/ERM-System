- [ ] **Step 1: Create the trigger SQL**

`backend/seed/58-template-immutability-trigger.sql`:

```sql
-- ============================================================
-- Description: Trigger that blocks content/variable_definitions
-- modification on template_versions once they leave DRAFT status.
-- Prevents tampering with REVIEW/APPROVED/DEPRECATED/ARCHIVED content.
-- Called BEFORE UPDATE on templates.template_versions.
-- ============================================================

CREATE OR REPLACE FUNCTION templates.fn_block_version_content_update()
RETURNS trigger
LANGUAGE plpgsql
AS $$
BEGIN
  IF OLD.status IN ('REVIEW', 'APPROVED', 'DEPRECATED', 'ARCHIVED') THEN
    IF NEW.content IS DISTINCT FROM OLD.content THEN
      RAISE EXCEPTION 'Cannot modify content of a version in status %', OLD.status
        USING HINT = 'Create a new version instead';
    END IF;
    IF NEW.variable_definitions IS DISTINCT FROM OLD.variable_definitions THEN
      RAISE EXCEPTION 'Cannot modify variable definitions of a version in status %', OLD.status
        USING HINT = 'Create a new version instead';
    END IF;
  END IF;

  -- Always allow status changes (handled by lifecycle service)
  -- Always allow effective_from/effective_until/retired_at changes
  -- Always allow approved_by/approved_at changes
  RETURN NEW;
END;
$$;

DROP TRIGGER IF EXISTS trg_block_version_content_update ON templates.template_versions;

CREATE TRIGGER trg_block_version_content_update
  BEFORE UPDATE ON templates.template_versions
  FOR EACH ROW
  EXECUTE FUNCTION templates.fn_block_version_content_update();
```

```typescript
// ============================================================
// Description: Repository for templates.template_versions CRUD
// and lifecycle data access. Supports version state machine
// transitions, effective date management, and soft deletes.
// All queries enforce soft-delete filtering via deleted_at IS NULL.
// ============================================================

import { AuditableRepository } from './auditable.repository';
import type { TemplateVersionRow } from '../shared/db-types';

export interface TemplateVersionRepository {
  findById(id: number): Promise<TemplateVersionRow | null>;
  findByTemplateId(templateId: number): Promise<TemplateVersionRow[]>;
  findByTemplateCode(code: string): Promise<TemplateVersionRow[]>;
  findByCodeAndVersion(code: string, version: string): Promise<TemplateVersionRow | null>;
  findApproved(templateId: number, now: Date): Promise<TemplateVersionRow | null>;
  findActive(templateId: number, now: Date): Promise<TemplateVersionRow[]>;
  create(data: CreateTemplateVersionInput, client?: any): Promise<TemplateVersionRow>;
  updateStatus(id: number, newStatus: string, userId: number, client?: any): Promise<TemplateVersionRow>;
  updateEffectiveDates(id: number, effectiveFrom: Date | null, effectiveUntil: Date | null, client?: any): Promise<TemplateVersionRow>;
  deprecateCurrentApproved(templateId: number, now: Date, client?: any): Promise<TemplateVersionRow | null>;
  existsByTemplateAndVersion(templateId: number, version: string): Promise<boolean>;
  softDelete(id: number): Promise<boolean>;
}

export interface CreateTemplateVersionInput {
  template_id: number;
  version: string;
  content: any;
  content_hash: string;
  variable_definitions: any;
  change_summary?: string;
  effective_from?: Date | null;
  effective_until?: Date | null;
  created_by: number;
}

export class TemplateVersionRepositoryImpl extends AuditableRepository implements TemplateVersionRepository {
  async findById(id: number): Promise<TemplateVersionRow | null> {
    const result = await this.query(
      `SELECT * FROM templates.template_versions WHERE id = $1`,
      [id],
    );
    return result.rows[0] || null;
  }

  async findByTemplateId(templateId: number): Promise<TemplateVersionRow[]> {
    const result = await this.query(
      `SELECT * FROM templates.template_versions WHERE template_id = $1 ORDER BY created_at DESC`,
      [templateId],
    );
    return result.rows;
  }

  async findByTemplateCode(code: string): Promise<TemplateVersionRow[]> {
    const result = await this.query(
      `SELECT tv.* FROM templates.template_versions tv
       JOIN templates.templates t ON t.id = tv.template_id
       WHERE t.code = $1 AND t.deleted_at IS NULL
       ORDER BY tv.created_at DESC`,
      [code],
    );
    return result.rows;
  }

  async findByCodeAndVersion(code: string, version: string): Promise<TemplateVersionRow | null> {
    const result = await this.query(
      `SELECT tv.* FROM templates.template_versions tv
       JOIN templates.templates t ON t.id = tv.template_id
       WHERE t.code = $1 AND tv.version = $2 AND t.deleted_at IS NULL`,
      [code, version],
    );
    return result.rows[0] || null;
  }

  async findApproved(templateId: number, now: Date): Promise<TemplateVersionRow | null> {
    const result = await this.query(
      `SELECT * FROM templates.template_versions
       WHERE template_id = $1 AND status = 'APPROVED'
         AND (effective_from IS NULL OR effective_from <= $2)
         AND (effective_until IS NULL OR effective_until > $2)
       LIMIT 1`,
      [templateId, now],
    );
    return result.rows[0] || null;
  }

  async findActive(templateId: number, now: Date): Promise<TemplateVersionRow[]> {
    const result = await this.query(
      `SELECT * FROM templates.template_versions
       WHERE template_id = $1
         AND status IN ('APPROVED', 'DEPRECATED')
         AND (effective_from IS NULL OR effective_from <= $2)
         AND (effective_until IS NULL OR effective_until > $2)
       ORDER BY
         CASE WHEN status = 'APPROVED' THEN 0 ELSE 1 END,
         created_at DESC`,
      [templateId, now],
    );
    return result.rows;
  }

  async create(data: CreateTemplateVersionInput, client?: any): Promise<TemplateVersionRow> {
    const result = await this.query(
      `INSERT INTO templates.template_versions
        (template_id, version, status, content, content_hash, variable_definitions,
         change_summary, effective_from, effective_until, created_by)
       VALUES ($1, $2, 'DRAFT', $3, $4, $5, $6, $7, $8, $9)
       RETURNING *`,
      [
        data.template_id, data.version, data.content, data.content_hash,
        data.variable_definitions, data.change_summary || null,
        data.effective_from || null, data.effective_until || null,
        data.created_by,
      ],
      client,
    );
    return result.rows[0];
  }

  async update(id: number, data: Record<string, any>, userId: number, client?: any): Promise<TemplateVersionRow> {
    const fields: string[] = [];
    const values: any[] = [];
    let idx = 1;

    if (data.content !== undefined) {
      fields.push(`content = $${idx++}`);
      values.push(data.content);
      fields.push(`content_hash = $${idx++}`);
      values.push(data.content_hash);
    }
    if (data.variable_definitions !== undefined) {
      fields.push(`variable_definitions = $${idx++}`);
      values.push(data.variable_definitions);
    }
    if (data.change_summary !== undefined) {
      fields.push(`change_summary = $${idx++}`);
      values.push(data.change_summary);
    }
    if (data.effective_from !== undefined) {
      fields.push(`effective_from = $${idx++}`);
      values.push(data.effective_from);
    }
    if (data.effective_until !== undefined) {
      fields.push(`effective_until = $${idx++}`);
      values.push(data.effective_until);
    }

    if (fields.length === 0) return (await this.findById(id))!;

    fields.push(`updated_by = $${idx++}`);
    values.push(userId);
    fields.push(`updated_at = NOW()`);

    const result = await this.query(
      `UPDATE templates.template_versions SET ${fields.join(', ')} WHERE id = $${idx} RETURNING *`,
      [...values, id],
      client,
    );
    if (!result.rows[0]) throw Object.assign(new Error('Template version not found'), { status: 404 });
    return result.rows[0];
  }

  async updateStatus(id: number, newStatus: string, userId: number, client?: any): Promise<TemplateVersionRow> {
    const result = await this.query(
      `UPDATE templates.template_versions
       SET status = $1, updated_by = $2, updated_at = NOW()
       WHERE id = $3
       RETURNING *`,
      [newStatus, userId, id],
      client,
    );
    if (!result.rows[0]) throw Object.assign(new Error('Template version not found'), { status: 404 });
    return result.rows[0];
  }

  async updateEffectiveDates(
    id: number, effectiveFrom: Date | null, effectiveUntil: Date | null, client?: any,
  ): Promise<TemplateVersionRow> {
    const result = await this.query(
      `UPDATE templates.template_versions
       SET effective_from = $1, effective_until = $2, updated_by = $3, updated_at = NOW()
       WHERE id = $4
       RETURNING *`,
      [effectiveFrom, effectiveUntil, this.getCurrentUserId(), id],
      client,
    );
    if (!result.rows[0]) throw Object.assign(new Error('Template version not found'), { status: 404 });
    return result.rows[0];
  }

  async deprecateCurrentApproved(templateId: number, now: Date, client?: any): Promise<TemplateVersionRow | null> {
    const result = await this.query(
      `UPDATE templates.template_versions
       SET status = 'DEPRECATED',
           effective_until = $1,
           updated_by = $2,
           updated_at = NOW()
       WHERE template_id = $3 AND status = 'APPROVED'
       RETURNING *`,
      [now, this.getCurrentUserId(), templateId],
      client,
    );
    return result.rows[0] || null;
  }

  async existsByTemplateAndVersion(templateId: number, version: string): Promise<boolean> {
    const result = await this.query(
      `SELECT 1 FROM templates.template_versions WHERE template_id = $1 AND version = $2`,
      [templateId, version],
    );
    return result.rows.length > 0;
  }

  async softDelete(id: number): Promise<boolean> {
    const meta = this.deleteMeta();
    const result = await this.query(
      `UPDATE templates.template_versions
       SET deleted_at = $1, deleted_by = $2
       WHERE id = $3 AND status = 'DRAFT'
       RETURNING id`,
      [meta.deleted_at, meta.deleted_by, id],
    );
    if (result.rows.length === 0) {
      throw Object.assign(
        new Error('Cannot delete version: not found or not in DRAFT status'),
        { status: 400 },
      );
    }
    return true;
  }
}
```

```typescript
// ============================================================
// Description: Repository for templates.template_approval_workflow
// multi-step approval tracking per version. Each version can
// have N sequential approval steps (by role or user).
// ============================================================

import { AuditableRepository } from './auditable.repository';

export interface ApprovalStepInput {
  template_version_id: number;
  step_order: number;
  approver_role: string;
  approver_id?: number | null;
}

export interface ApprovalStepRow {
  id: number;
  template_version_id: number;
  step_order: number;
  approver_role: string;
  approver_id: number | null;
  status: 'PENDING' | 'APPROVED' | 'REJECTED';
  comments: string | null;
  acted_by: number | null;
  acted_at: Date | null;
  created_at: Date;
}

export class TemplateApprovalWorkflowRepository extends AuditableRepository {
  async createSteps(steps: ApprovalStepInput[], client?: any): Promise<ApprovalStepRow[]> {
    if (steps.length === 0) return [];
    const values: any[] = [];
    const placeholders: string[] = [];
    let idx = 1;

    for (const step of steps) {
      placeholders.push(`($${idx++}, $${idx++}, $${idx++}, $${idx++})`);
      values.push(step.template_version_id, step.step_order, step.approver_role, step.approver_id ?? null);
    }

    const result = await this.query(
      `INSERT INTO templates.template_approval_workflow
        (template_version_id, step_order, approver_role, approver_id)
       VALUES ${placeholders.join(', ')}
       RETURNING *`,
      values,
      client,
    );
    return result.rows;
  }

  async findByVersionId(versionId: number): Promise<ApprovalStepRow[]> {
    const result = await this.query(
      `SELECT * FROM templates.template_approval_workflow
       WHERE template_version_id = $1
       ORDER BY step_order ASC`,
      [versionId],
    );
    return result.rows;
  }

  async approveStep(id: number, userId: number, comments?: string, client?: any): Promise<ApprovalStepRow> {
    const result = await this.query(
      `UPDATE templates.template_approval_workflow
       SET status = 'APPROVED', acted_by = $1, acted_at = NOW(), comments = COALESCE($2, comments)
       WHERE id = $3
       RETURNING *`,
      [userId, comments || null, id],
      client,
    );
    if (!result.rows[0]) throw Object.assign(new Error('Approval step not found'), { status: 404 });
    return result.rows[0];
  }

  async rejectStep(id: number, userId: number, comments?: string, client?: any): Promise<ApprovalStepRow> {
    const result = await this.query(
      `UPDATE templates.template_approval_workflow
       SET status = 'REJECTED', acted_by = $1, acted_at = NOW(), comments = COALESCE($2, comments)
       WHERE id = $3
       RETURNING *`,
      [userId, comments || null, id],
      client,
    );
    if (!result.rows[0]) throw Object.assign(new Error('Approval step not found'), { status: 404 });
    return result.rows[0];
  }

  async countPendingSteps(versionId: number): Promise<number> {
    const result = await this.query(
      `SELECT COUNT(*) as count FROM templates.template_approval_workflow
       WHERE template_version_id = $1 AND status = 'PENDING'`,
      [versionId],
    );
    return parseInt(result.rows[0].count, 10);
  }

  async countRejectedSteps(versionId: number): Promise<number> {
    const result = await this.query(
      `SELECT COUNT(*) as count FROM templates.template_approval_workflow
       WHERE template_version_id = $1 AND status = 'REJECTED'`,
      [versionId],
    );
    return parseInt(result.rows[0].count, 10);
  }

  async allStepsApproved(versionId: number): Promise<boolean> {
    const result = await this.query(
      `SELECT COUNT(*) as total,
              COUNT(*) FILTER (WHERE status = 'APPROVED') as approved
       FROM templates.template_approval_workflow
       WHERE template_version_id = $1`,
      [versionId],
    );
    const row = result.rows[0];
    return parseInt(row.total, 10) > 0 && parseInt(row.total, 10) === parseInt(row.approved, 10);
  }
}
```

### Task 5: VersionLifecycleService — State Machine

**Files:**

**Interfaces:**
- Consumes: `TemplateVersionRepository`, `TemplateVersionAuditRepository`, `TemplateApprovalWorkflowRepository`, `TemplateValidationService`
- Produces: `VersionLifecycleService` with:
  - `submit(code, version, user)` → submit for review (DRAFT→REVIEW)
  - `approve(code, version, user, comment?)` → approve (REVIEW→APPROVED)
  - `reject(code, version, user, reason)` → reject back to DRAFT (REVIEW→DRAFT)
  - `deprecate(code, version, user, reason?)` → deprecate (APPROVED→DEPRECATED)
  - `archive(code, version, user, reason?)` → archive (DEPRECATED→ARCHIVED)
  - `rollback(code, version, user, reason?)` → rollback (DEPRECATED→APPROVED)
  - `getTransitionHistory(code, version)` → audit entries
  - `getApprovalStatus(code, version)` → approval steps
  - `isVersionActive(code, version)` → boolean
  - `getActiveVersion(code)` → active version for generation

**State machine transitions:**

```
DRAFT → REVIEW      (submit)
REVIEW → APPROVED   (approve)
REVIEW → DRAFT      (reject)
APPROVED → DEPRECATED (deprecate)
DEPRECATED → ARCHIVED (archive)
DEPRECATED → APPROVED (rollback)
```

**Terminal states:** ARCHIVED has no outgoing transitions.

```typescript
// ============================================================
// Description: Version lifecycle state machine service for
// template versions. Handles all status transitions:
// DRAFT <-> REVIEW -> APPROVED -> DEPRECATED -> ARCHIVED.
// Enforces validation on submit, approval workflow on approve,
// and database integrity via partial unique index (one APPROVED).
// Every transition is recorded in the audit trail.
// ============================================================

import type { TemplateVersionRow } from '../shared/db-types';
import type { TemplateVersionRepository } from '../repositories/template-version.repository';
import type { TemplateVersionAuditRepository } from '../repositories/template-version-audit.repository';
import type { TemplateApprovalWorkflowRepository } from '../repositories/template-approval-workflow.repository';
import type { TemplateValidationService } from './template-validation.service';
import type { AuthUser } from '../shared/types';

type TransitionMap = Record<string, string[]>;

const ALLOWED_TRANSITIONS: TransitionMap = {
  DRAFT: ['REVIEW'],
  REVIEW: ['APPROVED', 'DRAFT'],
  APPROVED: ['DEPRECATED'],
  DEPRECATED: ['ARCHIVED', 'APPROVED'],
  ARCHIVED: [],
};

const TRANSITION_ACTION: Record<string, string> = {
  'DRAFT->REVIEW': 'SUBMITTED',
  'REVIEW->APPROVED': 'APPROVED',
  'REVIEW->DRAFT': 'REJECTED',
  'APPROVED->DEPRECATED': 'DEPRECATED',
  'DEPRECATED->ARCHIVED': 'ARCHIVED',
  'DEPRECATED->APPROVED': 'ROLLED_BACK',
};

export class VersionLifecycleService {
  constructor(
    private versionRepo: TemplateVersionRepository,
    private auditRepo: TemplateVersionAuditRepository,
    private approvalRepo: TemplateApprovalWorkflowRepository,
    private validationService: TemplateValidationService,
  ) {}

  async submit(templateCode: string, version: string, user: AuthUser, comment?: string): Promise<TemplateVersionRow> {
    const ver = await this.getVersionOrThrow(templateCode, version);
    this.assertTransition(ver.status, 'REVIEW');

    if (ver.status === 'DRAFT') {
      const validationResult = await this.validationService.validateVersion({
        template_id: ver.template_id,
        version: ver.version,
        status: ver.status,
        content: ver.content,
        content_hash: ver.content_hash,
        variable_definitions: ver.variable_definitions,
        change_summary: ver.change_summary || undefined,
        effective_from: ver.effective_from?.toISOString(),
        effective_until: ver.effective_until?.toISOString(),
        created_by: ver.created_by,
      });

      if (!validationResult.valid && validationResult.errors.length > 0) {
        throw Object.assign(
          new Error(`Cannot submit: validation failed (${validationResult.errors.length} error(s))`),
          { status: 400 },
        );
      }
    }

    const updated = await this.versionRepo.updateStatus(ver.id, 'REVIEW', user.id);
    await this.auditRepo.log({
      template_version_id: ver.id,
      action: 'SUBMITTED',
      actor_id: user.id,
      previous_status: ver.status,
      new_status: 'REVIEW',
      comment: comment || undefined,
    });

    return updated;
  }

  async approve(templateCode: string, version: string, user: AuthUser, comment?: string): Promise<TemplateVersionRow> {
    const ver = await this.getVersionOrThrow(templateCode, version);
    this.assertTransition(ver.status, 'APPROVED');

    if (ver.status === 'REVIEW') {
      const steps = await this.approvalRepo.findByVersionId(ver.id);
      if (steps.length > 0) {
        const allApproved = await this.approvalRepo.allStepsApproved(ver.id);
        if (!allApproved) {
          throw Object.assign(new Error('Approval workflow not complete: all steps must be APPROVED'), { status: 400 });
        }
      }
    }

    const updated = await this.versionRepo.updateStatus(ver.id, 'APPROVED', user.id);

    const previousApproved = await this.versionRepo.deprecateCurrentApproved(ver.template_id, new Date());
    if (previousApproved) {
      await this.auditRepo.log({
        template_version_id: previousApproved.id,
        action: 'SUPERSEDED',
        actor_id: user.id,
        previous_status: 'APPROVED',
        new_status: 'DEPRECATED',
        comment: `Superseded by version ${version}`,
      });
    }

    await this.auditRepo.log({
      template_version_id: ver.id,
      action: 'APPROVED',
      actor_id: user.id,
      previous_status: ver.status,
      new_status: 'APPROVED',
      comment: comment || undefined,
    });

    return updated;
  }

  async reject(templateCode: string, version: string, user: AuthUser, reason: string): Promise<TemplateVersionRow> {
    if (!reason || reason.trim().length === 0) {
      throw Object.assign(new Error('Rejection reason is required'), { status: 400 });
    }

    const ver = await this.getVersionOrThrow(templateCode, version);
    this.assertTransition(ver.status, 'DRAFT');

    const updated = await this.versionRepo.updateStatus(ver.id, 'DRAFT', user.id);

    await this.auditRepo.log({
      template_version_id: ver.id,
      action: 'REJECTED',
      actor_id: user.id,
      previous_status: ver.status,
      new_status: 'DRAFT',
      comment: reason,
    });

    return updated;
  }

  async deprecate(templateCode: string, version: string, user: AuthUser, reason?: string): Promise<TemplateVersionRow> {
    const ver = await this.getVersionOrThrow(templateCode, version);
    this.assertTransition(ver.status, 'DEPRECATED');

    const updated = await this.versionRepo.updateStatus(ver.id, 'DEPRECATED', user.id);

    await this.auditRepo.log({
      template_version_id: ver.id,
      action: 'DEPRECATED',
      actor_id: user.id,
      previous_status: ver.status,
      new_status: 'DEPRECATED',
      comment: reason || undefined,
    });

    return updated;
  }

  async archive(templateCode: string, version: string, user: AuthUser, reason?: string): Promise<TemplateVersionRow> {
    const ver = await this.getVersionOrThrow(templateCode, version);
    this.assertTransition(ver.status, 'ARCHIVED');

    const updated = await this.versionRepo.updateStatus(ver.id, 'ARCHIVED', user.id);

    await this.auditRepo.log({
      template_version_id: ver.id,
      action: 'ARCHIVED',
      actor_id: user.id,
      previous_status: ver.status,
      new_status: 'ARCHIVED',
      comment: reason || undefined,
    });

    return updated;
  }

  async rollback(templateCode: string, version: string, user: AuthUser, reason?: string): Promise<TemplateVersionRow> {
    const ver = await this.getVersionOrThrow(templateCode, version);
    this.assertTransition(ver.status, 'APPROVED');

    const updated = await this.versionRepo.updateStatus(ver.id, 'APPROVED', user.id);
    await this.versionRepo.updateEffectiveDates(ver.id, null, null);

    const currentActive = await this.versionRepo.findApproved(ver.template_id, new Date());
    if (currentActive && currentActive.id !== ver.id) {
      await this.versionRepo.deprecateCurrentApproved(ver.template_id, new Date());
      await this.auditRepo.log({
        template_version_id: currentActive.id,
        action: 'SUPERSEDED',
        actor_id: user.id,
        previous_status: 'APPROVED',
        new_status: 'DEPRECATED',
        comment: `Superseded by rollback to version ${version}`,
      });
    }

    await this.auditRepo.log({
      template_version_id: ver.id,
      action: 'ROLLED_BACK',
      actor_id: user.id,
      previous_status: 'DEPRECATED',
      new_status: 'APPROVED',
      comment: reason || undefined,
    });

    return updated;
  }

  async getTransitionHistory(templateCode: string, version: string): Promise<any[]> {
    const ver = await this.getVersionOrThrow(templateCode, version);
    return this.auditRepo.findByVersionId(ver.id);
  }

  async getApprovalStatus(templateCode: string, version: string): Promise<any[]> {
    const ver = await this.getVersionOrThrow(templateCode, version);
    return this.approvalRepo.findByVersionId(ver.id);
  }

  async isVersionActive(templateCode: string, version: string): Promise<boolean> {
    const ver = await this.getVersionOrThrow(templateCode, version);
    if (ver.status !== 'APPROVED') return false;

    const now = new Date();
    if (ver.effective_from && ver.effective_from > now) return false;
    if (ver.effective_until && ver.effective_until <= now) return false;

    return true;
  }

  async getActiveVersion(templateCode: string): Promise<TemplateVersionRow | null> {
    const versions = await this.versionRepo.findByTemplateCode(templateCode);
    const now = new Date();
    return versions.find(v =>
      v.status === 'APPROVED' &&
      (v.effective_from === null || v.effective_from <= now) &&
      (v.effective_until === null || v.effective_until > now)
    ) || null;
  }

  // ─── Private Helpers ──────────────────────────────────────

  private async getVersionOrThrow(templateCode: string, version: string): Promise<TemplateVersionRow> {
    const ver = await this.versionRepo.findByCodeAndVersion(templateCode, version);
    if (!ver) {
      throw Object.assign(new Error(`Version ${version} of template "${templateCode}" not found`), { status: 404 });
    }
    return ver;
  }

  private assertTransition(currentStatus: string, targetStatus: string): void {
    const allowed = ALLOWED_TRANSITIONS[currentStatus];
    if (!allowed || !allowed.includes(targetStatus)) {
      throw Object.assign(
        new Error(`Invalid transition: cannot move from ${currentStatus} to ${targetStatus}`),
        { status: 400 },
      );
    }
  }
}
```

```typescript
// ============================================================
// Description: Template version CRUD service. Creates, updates,
// deletes, and lists template versions. Content validation runs
// before updates and submissions. Only DRAFT versions are mutable.
// ============================================================

import { createHash } from 'crypto';
import type { TemplateVersionRow } from '../shared/db-types';
import type { TemplateVersionRepository } from '../repositories/template-version.repository';
import type { TemplateVersionAuditRepository } from '../repositories/template-version-audit.repository';
import type { TemplateValidationService } from './template-validation.service';
import type { AuthUser } from '../shared/types';
import { templateVersionSchema } from '../shared/template-validation.schemas';

export interface CreateVersionInput {
  version: string;
  content: any;
  variable_definitions?: any;
  change_summary?: string;
  effective_from?: Date | null;
  effective_until?: Date | null;
}

export interface UpdateVersionInput {
  content?: any;
  variable_definitions?: any;
  change_summary?: string;
  effective_from?: Date | null;
  effective_until?: Date | null;
}

export interface VersionDiff {
  version1: string;
  version2: string;
  contentChanges: Array<{ field: string; type: 'added' | 'removed' | 'changed'; oldValue?: any; newValue?: any }>;
  metadataChanged: boolean;
}

export class TemplateVersionService {
  constructor(
    private versionRepo: TemplateVersionRepository,
    private auditRepo: TemplateVersionAuditRepository,
    private validationService: TemplateValidationService,
  ) {}

  async create(templateCode: string, data: CreateVersionInput, user: AuthUser): Promise<TemplateVersionRow> {
    if (!/^\d+\.\d+\.\d+$/.test(data.version)) {
      throw Object.assign(new Error('Version must be semver (e.g. 1.0.0)'), { status: 400 });
    }

    const templateVersions = await this.versionRepo.findByTemplateCode(templateCode);
    if (templateVersions.length === 0) {
      throw Object.assign(new Error(`Template "${templateCode}" not found`), { status: 404 });
    }

    const templateId = templateVersions[0].template_id;

    const exists = await this.versionRepo.existsByTemplateAndVersion(templateId, data.version);
    if (exists) {
      throw Object.assign(new Error(`Version ${data.version} already exists for template "${templateCode}"`), { status: 409 });
    }

    const contentHash = createHash('sha256').update(JSON.stringify(data.content)).digest('hex');

    const created = await this.versionRepo.create({
      template_id: templateId,
      version: data.version,
      content: data.content,
      content_hash: contentHash,
      variable_definitions: data.variable_definitions || [],
      change_summary: data.change_summary,
      effective_from: data.effective_from || null,
      effective_until: data.effective_until || null,
      created_by: user.id,
    });

    await this.auditRepo.log({
      template_version_id: created.id,
      action: 'CREATED',
      actor_id: user.id,
      previous_status: null,
      new_status: 'DRAFT',
      comment: `Created version ${data.version}`,
    });

    return created;
  }

  async update(templateCode: string, version: string, data: UpdateVersionInput, user: AuthUser): Promise<TemplateVersionRow> {
    const ver = await this.getVersionOrThrow(templateCode, version);

    if (ver.status !== 'DRAFT') {
      throw Object.assign(new Error('Only DRAFT versions can be modified'), { status: 400 });
    }

    if (data.effective_from && data.effective_until && data.effective_until <= data.effective_from) {
      throw Object.assign(new Error('effective_until must be after effective_from'), { status: 400 });
    }

    const updateData: any = {};

    if (data.content !== undefined) {
      updateData.content = data.content;
      updateData.content_hash = createHash('sha256').update(JSON.stringify(data.content)).digest('hex');
    }
    if (data.variable_definitions !== undefined) {
      updateData.variable_definitions = data.variable_definitions;
    }
    if (data.change_summary !== undefined) {
      updateData.change_summary = data.change_summary;
    }
    if (data.effective_from !== undefined) {
      updateData.effective_from = data.effective_from;
    }
    if (data.effective_until !== undefined) {
      updateData.effective_until = data.effective_until;
    }

    // Run validation on the updated content
    if (data.content !== undefined || data.variable_definitions !== undefined) {
      const validationInput = {
        template_id: ver.template_id,
        version: ver.version,
        status: 'DRAFT',
        content: data.content ?? ver.content,
        content_hash: '',
        variable_definitions: data.variable_definitions ?? ver.variable_definitions,
        change_summary: data.change_summary || ver.change_summary || undefined,
        created_by: ver.created_by,
      };

      const validationResult = await this.validationService.validateVersion(validationInput as any);
      if (!validationResult.valid && validationResult.errors.length > 0) {
        throw Object.assign(
          new Error(`Validation failed: ${validationResult.errors.length} error(s)`),
          { status: 400 },
        );
      }
    }

    return this.versionRepo.update(ver.id, updateData, user.id);
  }

  async delete(templateCode: string, version: string, user: AuthUser): Promise<boolean> {
    const ver = await this.getVersionOrThrow(templateCode, version);

    if (ver.status !== 'DRAFT') {
      throw Object.assign(new Error('Only DRAFT versions can be deleted'), { status: 400 });
    }

    await this.auditRepo.log({
      template_version_id: ver.id,
      action: 'DELETED',
      actor_id: user.id,
      previous_status: ver.status,
      new_status: null,
      comment: `Deleted version ${version}`,
    });

    return this.versionRepo.softDelete(ver.id);
  }

  async getVersion(templateCode: string, version: string): Promise<TemplateVersionRow> {
    return this.getVersionOrThrow(templateCode, version);
  }

  async listVersions(templateCode: string): Promise<TemplateVersionRow[]> {
    return this.versionRepo.findByTemplateCode(templateCode);
  }

  async getActiveVersions(templateCode: string): Promise<TemplateVersionRow[]> {
    const templateVersions = await this.versionRepo.findByTemplateCode(templateCode);
    if (templateVersions.length === 0) {
      throw Object.assign(new Error(`Template "${templateCode}" not found`), { status: 404 });
    }
    return this.versionRepo.findActive(templateVersions[0].template_id, new Date());
  }

  async diffVersions(templateCode: string, version1: string, version2: string): Promise<VersionDiff> {
    const v1 = await this.getVersionOrThrow(templateCode, version1);
    const v2 = await this.getVersionOrThrow(templateCode, version2);

    const contentChanges: VersionDiff['contentChanges'] = [];

    // Compare content locales (ar, en)
    const allLocales = new Set([
      ...Object.keys(v1.content || {}),
      ...Object.keys(v2.content || {}),
    ]);

    for (const locale of allLocales) {
      const v1Locale = (v1.content as any)?.[locale] || {};
      const v2Locale = (v2.content as any)?.[locale] || {};

      const allFields = new Set([
        ...Object.keys(v1Locale),
        ...Object.keys(v2Locale),
      ]);

      for (const field of allFields) {
        if (!(field in v1Locale)) {
          contentChanges.push({ field: `${locale}.${field}`, type: 'added', newValue: v2Locale[field] });
        } else if (!(field in v2Locale)) {
          contentChanges.push({ field: `${locale}.${field}`, type: 'removed', oldValue: v1Locale[field] });
        } else if (JSON.stringify(v1Locale[field]) !== JSON.stringify(v2Locale[field])) {
          contentChanges.push({ field: `${locale}.${field}`, type: 'changed', oldValue: v1Locale[field], newValue: v2Locale[field] });
        }
      }
    }

    const metadataChanged = JSON.stringify(v1.variable_definitions) !== JSON.stringify(v2.variable_definitions);

    return {
      version1: version1,
      version2: version2,
      contentChanges,
      metadataChanged,
    };
  }

  // ─── Private Helpers ──────────────────────────────────────

  private async getVersionOrThrow(templateCode: string, version: string): Promise<TemplateVersionRow> {
    const ver = await this.versionRepo.findByCodeAndVersion(templateCode, version);
    if (!ver) {
      throw Object.assign(new Error(`Version ${version} of template "${templateCode}" not found`), { status: 404 });
    }
    return ver;
  }
}
```

```typescript
import { Router } from 'express';
```
